/**
 * "Analyze this page" server brain.
 *
 * Input  (POST JSON):  { action:"analyzePage", pageText:"<snapshot>", catalog:[{id,name,unit}] }
 * Output (JSON only):  the schema documented in buildAnalyzePrompt_ below.
 *
 * The extension (analyze.js) sends a PRIORITIZED Salesforce snapshot:
 *   line 1: "SOURCE: Salesforce Opportunity record"  (or a plain page)
 *   then:   "URL: …"
 *           "== DETECTED (rule-based) ==" — values the extension already parsed
 *           "== QUOTE INFORMATION ==", "== RENEWALS ==", "== SUBSCRIPTION INFORMATION ==",
 *           "== PRODUCTS ==", "== OPPORTUNITY INFORMATION ==", "== CONTACT ROLES ==",
 *           "== ACCOUNT DETAILS ==", then other sections and raw page text.
 * Noise sections (Stage/Field History, System Information, Activity, Chatter,
 * Notes, Files) are already stripped out before it reaches you.
 */
function analyzePage_(data) {
  var pageText = (data && data.pageText != null) ? String(data.pageText) : '';
  var catalog  = (data && Array.isArray(data.catalog)) ? data.catalog : [];
  if (!pageText.trim()) return { ok: true }; // nothing to analyze — extension falls back

  var prompt = buildAnalyzePrompt_(pageText, catalog);

  var raw;
  try {
    // Reuse your existing model helper here if you have one, e.g.:
    //   raw = callGemini_(prompt);
    raw = callGenerativeModel_(prompt);
  } catch (err) {
    return { ok: false, error: String(err) }; // extension falls back to rule-based
  }

  var obj = extractJson_(raw);
  if (!obj || typeof obj !== 'object') return { ok: false, error: 'model did not return JSON' };
  obj.ok = true;
  return obj;
}

/**
 * Pull the first JSON object out of the model's reply, tolerating stray prose or
 * ```json fences. Returns null if nothing parseable is found.
 */
function extractJson_(text) {
  if (text == null) return null;
  var s = String(text).trim();
  s = s.replace(/^```(?:json)?/i, '').replace(/```$/,'').trim();
  try { return JSON.parse(s); } catch (e) {}
  var start = s.indexOf('{'), end = s.lastIndexOf('}');
  if (start > -1 && end > start) {
    try { return JSON.parse(s.slice(start, end + 1)); } catch (e2) {}
  }
  return null;
}

/**
 * Reference model call (Google Gemini). REPLACE with your existing helper if you
 * already have one — it only needs to take the prompt and return the model's raw
 * text. Set GEMINI_API_KEY in Project Settings ▸ Script properties.
 * Must return a plain string.
 */
function callGenerativeModel_(prompt) {
  var key = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!key) throw new Error('GEMINI_API_KEY script property is not set');
  var url = 'https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=' + encodeURIComponent(key);
  var body = {
    contents: [{ role: 'user', parts: [{ text: prompt }] }],
    generationConfig: { temperature: 0, responseMimeType: 'application/json' }
  };
  var resp = UrlFetchApp.fetch(url, {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify(body),
    muteHttpExceptions: true
  });
  var code = resp.getResponseCode();
  if (code < 200 || code >= 300) throw new Error('model HTTP ' + code + ': ' + resp.getContentText());
  var parsed = JSON.parse(resp.getContentText());
  var cand = parsed && parsed.candidates && parsed.candidates[0];
  var part = cand && cand.content && cand.content.parts && cand.content.parts[0];
  return (part && part.text) ? part.text : '';
}
