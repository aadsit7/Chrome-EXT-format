  saveQuote, and registerUser are untouched.
- ONE handler, TWO brains. The POST body carries a "mode" field:
    • mode:"page"  (default) → a web-page / Salesforce snapshot → buildAnalyzePrompt_

  page brain until this is deployed, which still returns valid JSON.

/**
 * "Analyze this page" + "Speak to fill" server brain (both modes).
 *
 * Input  (POST JSON):
 *   { action:"analyzePage", mode:"page"|"voice", pageText:"<snapshot or transcript>",
 *     catalog:[{id,name,unit}] }
 *   mode is optional and defaults to "page".
 * Output (JSON only):  the schema documented in buildAnalyzePrompt_ (page) /
 *   buildVoicePrompt_ (voice) below. Voice returns the EXTENDED schema
 *   (customerType / dealType / extraDiscountPct / partnerMarginPct / premiumSupport).
 *
 * PAGE mode — the extension (analyze.js) sends a PRIORITIZED Salesforce snapshot:
 *   line 1: "SOURCE: Salesforce Opportunity record"  (or a plain page)
 *   then:   "URL: …"
 *           "== DETECTED (rule-based) ==" — values the extension already parsed
 *           "== QUOTE INFORMATION ==", "== RENEWALS ==", "== SUBSCRIPTION INFORMATION ==",
 *           "== PRODUCTS ==", "== OPPORTUNITY INFORMATION ==", "== CONTACT ROLES ==",
 *           "== ACCOUNT DETAILS ==", then other sections and raw page text.
 *   Noise sections (Stage/Field History, System Information, Activity, Chatter,
 *   Notes, Files) are already stripped out before it reaches you.
 *
 * VOICE mode — pageText is a short spoken transcript, e.g. "We are selling
 *   Application Workspace to Amazon with a fifteen percent discount. It's a new
 *   customer." Treat it as conversational quote dictation (see buildVoicePrompt_).
 */
function analyzePage_(data) {
  var pageText = (data && data.pageText != null) ? String(data.pageText) : '';
  var catalog  = (data && Array.isArray(data.catalog)) ? data.catalog : [];
  var mode     = (data && data.mode === 'voice') ? 'voice' : 'page';
  if (!pageText.trim()) return { ok: true }; // nothing to analyze — extension falls back

  var prompt = (mode === 'voice')
    ? buildVoicePrompt_(pageText, catalog)   // "Speak to fill" — spoken dictation
    : buildAnalyzePrompt_(pageText, catalog); // "Analyze this page" — page snapshot

  var raw;
  try {
    // Reuse your existing model helper here if you have one, e.g.:
    //   raw = callGemini_(prompt);
    raw = callGenerativeModel_(prompt);
  } catch (err) {
    return { ok: false, error: String(err) }; // extension falls back to rule-based
  }

  var obj = extractJson_(raw);
  if (!obj || typeof obj !== 'object') return { ok: false, error: 'model did not return JSON' };
  obj.ok = true;
  return obj;
}

/**
 * The rewritten AI prompt. It teaches the model the payload sections, the exact
 * Salesforce field names from the two real Opportunity pages, the precedence
 * rules for conflicting values, and the JSON schema to return — and forbids it
 * from inventing anything not present in the payload.
 */
function buildAnalyzePrompt_(pageText, catalog) {
  var catalogLines = (catalog || []).map(function (p) {
    return '  - id "' + p.id + '"  name "' + p.name + '"  (priced per ' + (p.unit === 'user' ? 'user' : 'endpoint') + ')';
  }).join('\n');

  return [
    'You extract quote fields from a web page snapshot for a sales-quote tool.',
    'The snapshot is usually a Salesforce Opportunity record (its first line says',
    '"SOURCE: Salesforce Opportunity record"). Return ONLY a single JSON object —',
    'no explanation, no markdown, no code fences. Use null (or omit the key) for',
    'anything not clearly present. NEVER invent, guess, or calculate values that',
    'are not in the snapshot.',
    '',
    'PRODUCT CATALOG (match products to these; return the catalog id OR the exact',
    'name — partial/substring name matches are fine):',
    catalogLines || '  (none provided)',
    '',
    'PAYLOAD LAYOUT: sections are marked "== SECTION NAME ==". A',
    '"== DETECTED (rule-based) ==" block lists values the extension already parsed',
    '— trust it, but prefer a clearer value from a named section when they',
    'disagree. Related lists appear as "Header | Header | …" rows.',
    '',
    'SALESFORCE FIELD NAMES you will see and how to map them:',
    '  • Account Name  → customer (the END CUSTOMER). This stays the customer even',
    '    when a partner/reseller is present.',
    '  • Partner/Reseller, or Subscription Type = "Reseller"  → partnerCompany',
    '    (the reseller). Do NOT put the reseller in "customer".',
    '  • Contact Roles related list → contactName / email / contactPhone. PREFER',
    '    the row whose "Primary" column is checked.',
    '  • Close Date, License Start Date, License Expiration Date, End Date,',
    '    Renewal Month Date, Start Date  → candidate dates (see precedence).',
    '  • Quote Expiration Date / Expiration Date (Quote Information) → quoteExpirationDate.',
    '  • Subscription Term  → termMonths. Salesforce stores it as a decimal, e.g.',
    '    "12.000000000000" means 12 months. A grid "Subscription Term" column is',
    '    the same field.',
    '  • Quantity (Renewals / Subscription Information), a Products related-list',
    '    Quantity column, Current Device Count, Maximum Device Count New → the',
    '    line quantity (see precedence).',
    '  • Endpoint Tier is a RANGE like "1,001 - 5,000" — it is context only, NEVER',
    '    a quantity.',
    '  • Product(s) (a semicolon list like "Right Click Tools;Insights Subscription")',
    '    and the Products related list → product lines. Match each against the catalog.',
    '  • ARR up for Renewal → the current annual price the customer pays today.',
    '',
    'STRIP "USD " / currency words and thousands commas from every number',
    '("USD 133,000.00" → 133000). Dates → ISO "YYYY-MM-DD".',
    '',
    'NEW BUSINESS vs RENEWAL:',
    '  • RENEWAL if Opportunity Record Type = "Renewal", or Subscription Type =',
    '    "Reseller", or you see Renewals/Subscription fields (ARR up for Renewal,',
    '    Renewal Month Date, License Expiration Date, Renewal ARR). Set isRenewal',
    '    true and fill "renewLines" (the products renewing at today\'s price).',
    '  • Otherwise NEW BUSINESS: set isRenewal false and fill "lines" from the',
    '    Products related list.',
    '',
    'PRECEDENCE when values conflict:',
    '  1. QUANTITY: Renewals/Subscription "Quantity" > Products related-list',
    '     Quantity > Account "Current Device Count". Never use Endpoint Tier.',
    '  2. RENEWAL DATE (renewalDate): License Expiration Date / End Date >',
    '     Renewal Month Date > Close Date. Close Date is a SALES FORECAST, not the',
    '     renewal date — use it only if nothing else exists.',
    '  3. CURRENT RENEWAL PRICE: use "ARR up for Renewal" as the current annual',
    '     price (currentAnnualPrice) for the renewing product(s).',
    '  4. CUSTOMER vs PARTNER: Account Name is the customer; the Partner/Reseller',
    '     (or the reseller implied by Subscription Type = "Reseller") is partnerCompany.',
    '',
    'RETURN EXACTLY THIS JSON SHAPE (include only keys you actually found):',
    '{',
    '  "isRenewal": true/false,',
    '  "customer": "end customer / Account Name",',
    '  "partnerCompany": "reseller company, if any",',
    '  "partnerEmail": "reseller email, if any",',
    '  "contactName": "primary contact name",',
    '  "email": "primary contact email",',
    '  "contactPhone": "primary contact phone",',
    '  "billToAddress": "", "shipToAddress": "",',
    '  "quoteExpirationDate": "YYYY-MM-DD",',
    '  "renewalDate": "YYYY-MM-DD",',
    '  "currency": "USD",',
    '  "termMonths": 12,',
    '  "lines": [ { "productId": "catalog id or name", "qty": 10000 } ],',
    '  "renewLines": [ { "product": "catalog id or name", "qty": 20000, "currentAnnualPrice": 133000 } ]',

    '',

    pageText,
    '--- PAGE SNAPSHOT END ---'
  ].join('\n');
}

/**
 * The "Speak to fill" (voice) prompt. Here pageText is a short SPOKEN transcript,
 * not a page — a rep dictating a quote in natural language, in any word order.
 * It teaches the model how people actually talk ("selling X to Y", "through
 * partner Z", spelled-out numbers, discount vs margin), the same strict-JSON /
 * never-invent rules as page mode, and the EXTENDED schema the extension applies
 * (customerType / dealType / extraDiscountPct / partnerMarginPct / premiumSupport
 * plus customer / partnerCompany / lines / renewLines / term / contact).
 */
function buildVoicePrompt_(pageText, catalog) {
  var catalogLines = (catalog || []).map(function (p) {
    return '  - id "' + p.id + '"  name "' + p.name + '"  (priced per ' + (p.unit === 'user' ? 'user' : 'endpoint') + ')';
  }).join('\n');

  return [
    'You extract sales-quote fields from a short SPOKEN transcript. A sales rep is',
    'dictating a quote out loud, in natural language and ANY word order. Return',
    'ONLY a single JSON object — no explanation, no markdown, no code fences. Use',
    'null (or omit the key) for anything not clearly said. NEVER invent, guess, or',
    'calculate values that were not spoken. Speech may be slightly misheard; map',
    'only what is clearly there.',
    '',
    'PRODUCT CATALOG (match spoken products to these; return the catalog id OR the',
    'exact name — partial/substring matches are fine, e.g. "app workspace" → the',
    'Application Workspace id):',
    catalogLines || '  (none provided)',
    '',
    'HOW PEOPLE SAY IT — map these patterns (they can appear in any order):',
    '  • "selling X to Y", "sell X to Y", "quoting X to Y", "quote for Y", "deal',
    '    with Y", "for Y"  →  Y is the CUSTOMER (customer) and X is the PRODUCT.',
    '    The customer is the company being sold to, NOT the product and NOT the rep.',
    '  • "through partner Z", "via Z", "with partner Z", "reseller Z", "sell through',
    '    Z"  →  Z is the reseller (partnerCompany). A partner/reseller is NOT the',
    '    customer.',
    '  • "N percent discount", "discount of N percent", "N% off", "a N percent',
    '    sweetener" — in ANY word order  →  extraDiscountPct = N. BUT if the word',
    '    "margin" is used ("N percent margin", "partner margin of N", "margin of N',
    '    percent")  →  partnerMarginPct = N instead (and that implies a partner',
    '    deal). "annual increase / uplift / escalation of N percent" is neither —',
    '    ignore it for these two fields.',
    '  • "new customer", "net new", "brand new logo"  →  customerType = "new".',
    '    "current customer", "existing customer", "current account"  →  customerType',
    '    = "current".',
    '  • "renewal" / "renewing"  →  dealType = "ren". "add-on" / "upsell" / "expansion"',
    '    →  dealType = "addon". "add-on and renewal" / "add-on plus renewal"  →',
    '    dealType = "addonren". (dealType only applies to a current customer.)',
    '  • "premium support", "with support", "add support"  →  premiumSupport = true.',
    '    "no support", "without premium support"  →  premiumSupport = false.',
    '  • "N year term", "N years", "a term of N years", "N month term"  →  termMonths',
    '    (years × 12). "two year term" = 24.',
    '  • a quantity near a product ("2,500 endpoints", "500 users", "needs 2,500 of',
    '    them")  →  that product line\'s qty.',
    '',
    'SPELLED-OUT NUMBERS: convert every number word to a figure — "fifteen" → 15,',
    '"twenty five hundred" / "two thousand five hundred" → 2500, "three" → 3,',
    '"a couple thousand" → 2000. Strip thousands commas ("2,500" → 2500). Percentages',
    'stay as the plain number (15, not 0.15).',
    '',
    'NEW BUSINESS vs RENEWAL: if the rep says "renewal" set isRenewal true and put',
    'the products under "renewLines"; otherwise set isRenewal false and put them',
    'under "lines". Use one or the other, not both.',
    '',
    'RETURN EXACTLY THIS JSON SHAPE (include only keys you actually heard):',
    '{',
    '  "isRenewal": true/false,',
    '  "customer": "the company being sold to",',
    '  "partnerCompany": "reseller / partner, if any",',
    '  "customerType": "new" | "current",',
    '  "dealType": "addon" | "ren" | "addonren",',
    '  "extraDiscountPct": 15,',
    '  "partnerMarginPct": 20,',
    '  "premiumSupport": true/false,',
    '  "contactName": "person named, if any",',
    '  "email": "email, if spelled out (\\"jane at acme dot com\\" → jane@acme.com)",',
    '  "termMonths": 24,',
    '  "lines": [ { "productId": "catalog id or name", "qty": 2500 } ],',
    '  "renewLines": [ { "product": "catalog id or name", "qty": 20000, "currentAnnualPrice": 133000 } ]',
    '}',
    '',
    'EXAMPLE — "We are selling Application Workspace to Amazon with a fifteen',
    'percent discount. It\'s a new customer." →',
    '{ "isRenewal": false, "customer": "Amazon", "customerType": "new",',
    '  "extraDiscountPct": 15, "lines": [ { "productId": "aw" } ] }',
    'Note the customer is Amazon (the company), the product is Application Workspace,',
    'the 15% is the extra discount (no "margin" was said), and it is net new.',
    '',
    '--- SPOKEN TRANSCRIPT START ---',
    pageText,
    '--- SPOKEN TRANSCRIPT END ---'
  ].join('\n');
}

/**
 * Pull the first JSON object out of the model's reply, tolerating stray prose or
 * ```json fences. Returns null if nothing parseable is found.
 */
function extractJson_(text) {
  if (text == null) return null;
  var s = String(text).trim();
  s = s.replace(/^```(?:json)?/i, '').replace(/```$/,'').trim();
  try { return JSON.parse(s); } catch (e) {}
  var start = s.indexOf('{'), end = s.lastIndexOf('}');
  if (start > -1 && end > start) {
    try { return JSON.parse(s.slice(start, end + 1)); } catch (e2) {}
  }
  return null;
}

/**
 * Reference model call (Google Gemini). REPLACE with your existing helper if you
 * already have one — it only needs to take the prompt and return the model's raw
 * text. Set GEMINI_API_KEY in Project Settings ▸ Script properties.
 * Must return a plain string.
 */
function callGenerativeModel_(prompt) {
  var key = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!key) throw new Error('GEMINI_API_KEY script property is not set');
  var url = 'https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=' + encodeURIComponent(key);
  var body = {
    contents: [{ role: 'user', parts: [{ text: prompt }] }],
    generationConfig: { temperature: 0, responseMimeType: 'application/json' }
  };
  var resp = UrlFetchApp.fetch(url, {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify(body),
    muteHttpExceptions: true
  });
  var code = resp.getResponseCode();
  if (code < 200 || code >= 300) throw new Error('model HTTP ' + code + ': ' + resp.getContentText());
  var parsed = JSON.parse(resp.getContentText());
  var cand = parsed && parsed.candidates && parsed.candidates[0];
  var part = cand && cand.content && cand.content.parts && cand.content.parts[0];
  return (part && part.text) ? part.text : '';
}

/**
 * "Look up address" server brain (extension v3.5+, Billing details).
 *
 * Input  (POST JSON):  { action:"addressLookup", company:"<company name>" }
 * Output (JSON only):  { ok:true, address:"Street\nCity, State ZIP\nCountry" }
 *                      address:"" when the company can't be identified.
 *
 * The extension asks THIS handler first for the company's corporate mailing /
 * headquarters address (AI-curated, well formatted). If this deployment doesn't
 * have the handler yet (doPost answers ok:false "unknown action") or the call
 * fails, the extension automatically falls back to its built-in keyless
 * OpenStreetMap lookup — so nothing breaks either way; pasting this just
 * upgrades the answer quality. The filled address is always labeled
 * "Auto-filled — please verify" in the panel.
 */
function addressLookup_(data) {
  var company = (data && data.company != null) ? String(data.company).trim() : '';
  if (!company) return { ok: true, address: '' };
  var prompt =
    'You are looking up the corporate mailing address (headquarters) of a company ' +
    'for a sales quote "Bill To" block.\n' +
    'Company: "' + company + '"\n\n' +
    'Return ONLY a JSON object, no prose, no markdown fences, exactly:\n' +
    '{ "address": "<street address>\\n<city, state/region postal code>\\n<country>" }\n' +
    'Rules:\n' +
    '- The input is ALWAYS the name of a company or organization — never a place, ' +
    'a river, a person, or a common word. "Amazon" means Amazon.com, Inc.; "Apple" ' +
    'means Apple Inc.; "Shell" means Shell plc.\n' +
    '- If several companies share the name, choose the most widely known one.\n' +
    '- Use the company\'s primary/global corporate headquarters mailing address — ' +
    'the address a supplier would put on an invoice (street and number, suite if ' +
    'applicable, city, state/region, postal code, country).\n' +
    '- 2 or 3 lines joined with \\n; do NOT repeat the company name in the address.\n' +
    '- If you are not confident you know this company\'s headquarters address, return ' +
    '{ "address": "" } instead of guessing a made-up street address.';
  var raw;
  try {
    // Reuse your existing model helper here if you have one, e.g. callGemini_(prompt);
    raw = callGenerativeModel_(prompt);
  } catch (err) {
    return { ok: false, error: String(err) }; // extension falls back to OpenStreetMap
  }
  var parsed = extractJson_(raw);
  var address = (parsed && parsed.address != null) ? String(parsed.address).trim() : '';
  return { ok: true, address: address };
}
